cmake_minimum_required(VERSION 3.20)

# PS4 build of EoSD-portable. Configure with:
#   source /opt/pacbrew/ps4/openorbis/ps4vars.sh
#   openorbis-cmake -G Ninja -S ps4/th06 -B build/th06
project(th06_ps4 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(TH06_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../src/th06)
set(PS4_COMMON_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../common)

set(TH06_SOURCES
    AnmManager.cpp AsciiManager.cpp BombData.cpp BulletData.cpp BulletManager.cpp
    Chain.cpp Controller.cpp EclManager.cpp EffectManager.cpp Ending.cpp
    EnemyEclInstr.cpp EnemyManager.cpp FileSystem.cpp GameErrorContext.cpp
    GameManager.cpp GameWindow.cpp Gui.cpp ItemManager.cpp main.cpp MainMenu.cpp
    MidiOutput.cpp MusicRoom.cpp Player.cpp ReplayManager.cpp ResultScreen.cpp
    Rng.cpp ScreenEffect.cpp SoundPlayer.cpp Stage.cpp Supervisor.cpp
    TextHelper.cpp ZunMath.cpp utils.cpp ZunTimer.cpp
    graphics/GLFunc.cpp graphics/FixedFunctionGL.cpp graphics/WebGL.cpp
    graphics/Software.cpp graphics/SoftwareRaster.cpp graphics/Gnm.cpp
    pbg3/FileAbstraction.cpp pbg3/IPbg3Parser.cpp pbg3/Pbg3Archive.cpp
    pbg3/Pbg3Parser.cpp
    thirdparty/sjis_converter.cpp
    midi/MidiDefault.cpp
)
list(TRANSFORM TH06_SOURCES PREPEND ${TH06_DIR}/src/)

# Same job as embedResource() in premake5.lua: turn the shaders into C arrays.
set(EMBEDS_HPP ${CMAKE_CURRENT_BINARY_DIR}/autogenerated/Embeds.hpp)
add_custom_command(
    OUTPUT ${EMBEDS_HPP}
    COMMAND ${CMAKE_COMMAND}
        -DOUT=${EMBEDS_HPP}
        -DVERT=${TH06_DIR}/src/graphics/ff.vert
        -DFRAG=${TH06_DIR}/src/graphics/ff.frag
        -P ${CMAKE_CURRENT_SOURCE_DIR}/embed.cmake
    DEPENDS ${TH06_DIR}/src/graphics/ff.vert ${TH06_DIR}/src/graphics/ff.frag
            ${CMAKE_CURRENT_SOURCE_DIR}/embed.cmake
)

add_executable(th06 ${TH06_SOURCES} ${EMBEDS_HPP}
    ${PS4_COMMON_DIR}/ps4_platform.cpp
    ${PS4_COMMON_DIR}/ps4_keyboard.cpp
    ${PS4_COMMON_DIR}/thpatch/thpatch.cpp
    ${PS4_COMMON_DIR}/thpatch/json.cpp
)
target_include_directories(th06 PRIVATE
    ${TH06_DIR}/src ${CMAKE_CURRENT_BINARY_DIR} ${PS4_COMMON_DIR} ${PS4_COMMON_DIR}/thpatch)
target_compile_definitions(th06 PRIVATE
    NDEBUG TH_PS4 TH_THPATCH TH_PS4_GAME_DIR="/data/touhou/th06"
)
target_compile_options(th06 PRIVATE
    # -march=btver2: the PS4's Jaguar cores (matters for the software rasterizer).
    -O2 -march=btver2 -Wno-gnu-anonymous-struct -Wno-unused-command-line-argument
    "SHELL:-include ${PS4_COMMON_DIR}/ps4_compat.hpp"
)

find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2_PS4 REQUIRED IMPORTED_TARGET GLOBAL sdl2 SDL2_image SDL2_ttf)
target_link_libraries(th06 PRIVATE ${SDL2_PS4_STATIC_LDFLAGS} SceVideoOut SceUserService SceSysmodule)

# Lets read-only files fall back to /app0/assets (see __wrap_fopen in ps4_platform.cpp).
target_link_options(th06 PRIVATE "LINKER:--wrap=fopen" "LINKER:--wrap=mkdir")
# Piglet (the system's OpenGL ES) only initializes in apps launched as mini apps (category
# "gde"): as a big app ("gd") sceCompositorInit fails. Mini apps can't record gameplay, but
# every Piglet homebrew (RetroArch included) is one. TH_PS4_BIG_APP keeps the big-app
# groundwork (direct-memory heap) for a future renderer that doesn't use Piglet.
option(TH_PS4_BIG_APP "Package as a big app (needs a non-Piglet renderer)" OFF)
if(TH_PS4_BIG_APP)
    set(TH_PS4_CATEGORY "gd")
    # The PS4's own GPU through GNM. Off falls back to the software rasterizer, which also
    # stays as the automatic fallback if the GPU backend fails to start.
    option(TH_PS4_GNM "Render with the GPU (GNM) in big-app builds" ON)
    target_compile_definitions(th06 PRIVATE TH_PS4_BIG_APP)
    if(TH_PS4_GNM)
        # The GPU draws at the display's resolution, like the Piglet build does.
        target_compile_definitions(th06 PRIVATE GAME_WINDOW_WIDTH_REAL=1920 GAME_WINDOW_HEIGHT_REAL=1080)
    else()
        # Software rasterizer at the native 640x480; ps4_videoout.cpp scales it to the screen.
        target_compile_definitions(th06 PRIVATE GAME_WINDOW_WIDTH_REAL=640 GAME_WINDOW_HEIGHT_REAL=480)
    endif()
    target_sources(th06 PRIVATE ${PS4_COMMON_DIR}/ps4_malloc.cpp ${PS4_COMMON_DIR}/ps4_videoout.cpp)
    foreach(fn malloc free calloc realloc aligned_alloc memalign posix_memalign)
        target_link_options(th06 PRIVATE "LINKER:--wrap=${fn}")
    endforeach()
    if(TH_PS4_GNM)
        set(OPENGNM_STACK "$ENV{HOME}/opengnm-stack" CACHE PATH "PS4-OpenGNM stack checkout")
        set(PSBC "$ENV{HOME}/old-psbc/zig-out/bin/psbc" CACHE FILEPATH "PS4 shader binary compiler")
        set(EX ${OPENGNM_STACK}/freegnm-examples)
        target_compile_definitions(th06 PRIVATE TH_PS4_GNM)
        target_sources(th06 PRIVATE
            ${EX}/shared/src/u/fs.c ${EX}/shared/src/u/utility.c ${EX}/shared/src/displayctx.c
            ${EX}/shared/src/memalloc.c ${EX}/shared/src/misc.c)
        set_source_files_properties(
            ${EX}/shared/src/u/fs.c ${EX}/shared/src/u/utility.c ${EX}/shared/src/displayctx.c
            ${EX}/shared/src/memalloc.c ${EX}/shared/src/misc.c
            PROPERTIES COMPILE_OPTIONS "-std=c11;-D_BSD_SOURCE;-DUSE_OPENGNM")
        target_include_directories(th06 BEFORE PRIVATE
            ${OPENGNM_STACK}/opengnm/include ${OPENGNM_STACK}/opengnm/include/gnm ${EX}/shared/src)
        target_link_libraries(th06 PRIVATE ${OPENGNM_STACK}/opengnm/libopengnm.a SceGnmDriver SceSystemService)

        # GLSL -> SPIR-V (glslc) -> PS4 shader binary (psbc). opengnm-psbc can't do this:
        # compiling a shader on its own, it replaces every texture and uniform buffer with a
        # zero descriptor, and the GPU faults reading address 0.
        set(TH_SHADERS)
        foreach(pair "th_gnm.vert:vertex:${TH06_DIR}/src/graphics"
                     "th_gnm.frag:fragment:${TH06_DIR}/src/graphics"
                     "clear.frag:fragment:${EX}/triangle/assets/misc")
            string(REPLACE ":" ";" parts ${pair})
            list(GET parts 0 name)
            list(GET parts 1 stage)
            list(GET parts 2 dir)
            set(out ${CMAKE_BINARY_DIR}/pkg/assets/misc/${name}.sb)
            add_custom_command(OUTPUT ${out}
                COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/pkg/assets/misc
                COMMAND glslc -fshader-stage=${stage} ${dir}/${name}.glsl -o ${CMAKE_BINARY_DIR}/${name}.spv
                COMMAND ${PSBC} -s ${stage} ${CMAKE_BINARY_DIR}/${name}.spv ${out}
                DEPENDS ${dir}/${name}.glsl)
            list(APPEND TH_SHADERS ${out})
        endforeach()
        add_custom_target(th06_shaders ALL DEPENDS ${TH_SHADERS})
    endif()
else()
    set(TH_PS4_CATEGORY "gde")
    # Render at 1080p through Piglet; GameWindow.hpp pillarboxes the 4:3 playfield itself.
    target_compile_definitions(th06 PRIVATE GAME_WINDOW_WIDTH_REAL=1920 GAME_WINDOW_HEIGHT_REAL=1080)
endif()

set(TH_GAME_ID th06)
include(${PS4_COMMON_DIR}/pkg.cmake)
th_prepare_pkg_dir(${CMAKE_BINARY_DIR}/pkg ${CMAKE_CURRENT_SOURCE_DIR}/pkg
    紅魔郷CM.DAT 紅魔郷ED.DAT 紅魔郷IN.DAT 紅魔郷MD.DAT 紅魔郷ST.DAT 紅魔郷TL.DAT bgm)

include(${PS4_COMMON_DIR}/self.cmake)
if(TH_PS4_BIG_APP)
    th_add_self(th06 GAME)
else()
    th_add_self(th06 MINIAPP)
endif()
add_pkg(th06 ${CMAKE_BINARY_DIR}/pkg "THSP00006" "Touhou 6 - The Embodiment of Scarlet Devil" "01.00" 0
        ${TH_PS4_CATEGORY})
if(TH_PS4_BIG_APP AND TH_PS4_GNM)
    add_dependencies(th06_pkg th06_shaders)
endif()
